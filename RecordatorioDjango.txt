Conexión MySQL 
- Creamos la base de datos con MySQL
    CREATE DATABASE elements CHARACTER SET UTF8MB4 COLLATE utf8mb4_general_ci;

-Usuarios
   --- Crea los Usuarios
   CREATE USER 'arman'@'localhost' IDENTIFIED BY 'password';
   -- Da privilegios
   GRANT ALL PRIVILEGES ON elements.* TO 'arman'@'localhost';
   --Recarga privilegios
   FLUSH PRIVILEGES;

Primero hacemos las migraciones
